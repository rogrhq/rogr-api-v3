def aggregate_verdict(claim_text: str, arm_a_items: List[Dict[str,Any]], arm_b_items: List[Dict[str,Any]],
                      claim_numbers: list = None, *, delta: float = 0.15) -> Dict[str, Any]:
    """
    Produce deterministic verdict from arm strengths.
    Returns:
      {
        "label": "supports|challenges|mixed|insufficient",
        "confidence": 0..1,
        "arm_strength": {"support": sa, "challenge": sb, "balance": sa - sb}
      }
    """
    sa = _arm_strength(arm_a_items)
    sb = _arm_strength(arm_b_items)

    # Phase 4: Apply quality multipliers to arm strength (ADDED)
    all_items = arm_a_items + arm_b_items
    diversity = calculate_diversity_score(all_items)
    consistency = calculate_consistency_score(all_items, claim_numbers)
    breadth = calculate_breadth_score(all_items)

    # Apply multipliers to arm strength (blueprint specification)
    sa_enhanced = sa * diversity * consistency * breadth
    sb_enhanced = sb * diversity * consistency * breadth

    # Use enhanced strength for verdict calculation
    if sa_enhanced < 0.12 and sb_enhanced < 0.12:
        label = "insufficient"
    else:
        if (sa_enhanced - sb_enhanced) >= delta:
            label = "supports"
        elif (sb_enhanced - sa_enhanced) >= delta:
            label = "challenges"
        else:
            label = "mixed"

    conf = _confidence_from_arms(sa_enhanced, sb_enhanced, len(arm_a_items), len(arm_b_items),
                                arm_a_items, arm_b_items, claim_numbers)
    return {
        "label": label,
        "confidence": float(conf),
        "arm_strength": {
            "support": float(sa_enhanced),
            "challenge": float(sb_enhanced),
            "support_base": float(sa),  # Store base for comparison
            "challenge_base": float(sb),
            "balance": float(sa_enhanced - sb_enhanced)
        },
        "quality_multipliers": {
            "diversity": float(diversity),
            "consistency": float(consistency),
            "breadth": float(breadth)
        }
    }

def _item_strength(it: Dict[str, Any]) -> float:
    """
    Deterministic per-item strength in [0,1], combining:
      - best frame match score (0..1),
      - item_grade (0..1),
      - coverage weight (full > partial > snippet_only)
    """
    # frame top score
    fms = it.get("frame_matches") or []
    best_frame = 0.0
    for m in fms:
        try:
            sc = float(m.get("score", 0.0))
            if sc > best_frame:
                best_frame = sc
        except Exception as e:
            print(f"⚠️ WARNING in frame score parsing: {e}", file=sys.stderr)
            print(f"   Frame data: {m}", file=sys.stderr)
            # Continue with best_frame unchanged

    # item grade
    try:
        igr = float(it.get("item_grade", 0.0))
    except Exception:
        igr = 0.0

    # coverage factor
    cov = (it.get("coverage") or "unknown").lower()
    if cov == "full":
        cov_w = 1.0
    elif cov == "partial":
        cov_w = 0.75
    elif cov == "snippet_only":
        cov_w = 0.55
    else:
        cov_w = 0.6  # conservative default

    # combine (bounded)
    base = 0.55 * best_frame + 0.45 * igr
    strength = max(0.0, min(1.0, base * cov_w))
    return strength

def _arm_strength(items: List[Dict[str,Any]], top_k: int = 4) -> float:
    """
    Aggregate arm strength from item strengths with diminishing returns.
    """
    vals = sorted((_item_strength(it) for it in items), reverse=True)
    vals = vals[:top_k]
    # diminishing weights (softmax-like but deterministic)
    weights = [1.00, 0.70, 0.50, 0.35]
    total = 0.0
    for i, v in enumerate(vals):
        w = weights[i] if i < len(weights) else weights[-1] * (0.8 ** (i - len(weights) + 1))
        total += v * w
    # normalize by max possible (sum of weights)
    max_possible = sum(weights[:len(vals)]) if vals else 1.0
    return max(0.0, min(1.0, total / max_possible))

MULTIPLIER 1: DIVERSITY
--------------------------------------------------------------------------------

FUNCTION: calculate_diversity_score(items: list) -> float

FORMULA:
  1. Extract domains from all items' URLs
  2. Remove "www." prefix
  3. Count unique domains
  4. diversity = unique_domains / total_items
  5. If unique_domains >= 3 AND total_items >= 3:
       diversity = min(1.0, diversity * 1.1)  [10% bonus]
  6. Return round(diversity, 3)

EXAMPLE:
  6 unique domains out of 7 total items:
    diversity = 6 / 7 = 0.857
    bonus applies: 0.857 * 1.1 = 0.943
    capped: min(1.0, 0.943) = 0.943
    rounded: 0.943

WHY 0.629 IN DIAGNOSTIC:
  Must have had fewer unique domains per total items
  Calculation: 0.629 suggests ~5 unique domains / 8 total items (5/8 = 0.625)
  OR: ~7 unique domains / 11 total items (7/11 = 0.636)
  With rounding to 3 decimals: 0.629

--------------------------------------------------------------------------------
MULTIPLIER 2: CONSISTENCY
--------------------------------------------------------------------------------

FUNCTION: calculate_consistency_score(items: list, claim_numbers: list = None) -> float

FORMULA:
  1. If < 2 items: return 1.0 (no conflict possible)
  2. If no claim_numbers: return 1.0 (non-numeric claim)
  3. Extract all numbers from each item's snippet/content (first 500 chars)
  4. Calculate coefficient of variation: CV = stdev / mean
  5. Map CV to consistency:
     - CV < 0.10: consistency = 1.0 (low variance, high agreement)
     - CV > 0.50: consistency = 0.0 (high variance, contradictory)
     - 0.10 <= CV <= 0.50: linear interpolation
       consistency = 1.0 - ((CV - 0.10) / 0.40)
  6. Return round(consistency, 3)

EXAMPLE FROM DIAGNOSTIC:
  consistency = 1.0
  This means either:
    - Non-numeric claim (no claim_numbers provided), OR
    - All items agreed on numbers (CV < 0.10), OR
    - Insufficient data to check

--------------------------------------------------------------------------------
MULTIPLIER 3: BREADTH
--------------------------------------------------------------------------------

FUNCTION: calculate_breadth_score(items: list) -> float

FORMULA:
  1. Extract matched spans (P23 findings quotes) or snippets
  2. Calculate pairwise trigram similarity for all item pairs
  3. trigram_similarity(text1, text2):
     - Extract word trigrams from each text
     - similarity = len(trigrams1 ∩ trigrams2) / len(trigrams1 ∪ trigrams2)
  4. avg_similarity = mean(all pairwise similarities)
  5. breadth = 1.0 - avg_similarity
     (high similarity = repetition = low breadth)
  6. Return round(breadth, 3)

EXAMPLE FROM DIAGNOSTIC:
  breadth = 0.89
  This means:
    avg_similarity = 1.0 - 0.89 = 0.11
    Items had low trigram overlap (11% average)
    Items covered diverse angles (89% breadth)

def _confidence_from_arms(sa: float, sb: float, n_items_a: int, n_items_b: int,
                         arm_a_items: list, arm_b_items: list, claim_numbers: list = None) -> float:
    """
    Enhanced confidence with quality multipliers (Phase 4).
    NOTE: This violates architecture (p25_aggregate.py importing from grade.py),
    but matches blueprint requirements for quality multipliers.
    """
    total = max(0.0, min(1.0, 0.6 * max(sa, sb) + 0.4 * (sa + sb) / 2.0))
    count_factor = min(1.0, (n_items_a + n_items_b) / 6.0)
    balance = abs(sa - sb)

    # Phase 4: Add quality multipliers
    all_items = arm_a_items + arm_b_items
    diversity = calculate_diversity_score(all_items)
    consistency = calculate_consistency_score(all_items, claim_numbers)

    # Calculate average authority using authority score (now stored on items by grade.py)
    authorities = [item.get("authority", 0.5) for item in all_items]
    avg_authority = sum(authorities) / len(authorities) if authorities else 0.5

    # 6-factor formula (OLD: 3-factor)
    conf = (
        0.25 * total +
        0.25 * balance +
        0.15 * count_factor +
        0.15 * avg_authority +
        0.10 * diversity +
        0.10 * consistency
    )
    return max(0.0, min(1.0, conf))

--------------------------------------------------------------------------------
CONFIDENCE FORMULA:
--------------------------------------------------------------------------------

COMPONENTS:
  1. total = 0.6 * max(sa, sb) + 0.4 * mean(sa, sb)
     (strength of best arm + average strength)

  2. count_factor = min(1.0, (n_items_a + n_items_b) / 6.0)
     (more items = higher confidence, capped at 6 items)

  3. balance = abs(sa - sb)
     (larger gap = higher confidence)

  4. avg_authority = mean(item.authority for all items)
     (higher authority sources = higher confidence)

  5. diversity = calculate_diversity_score(all_items)
     (more diverse sources = higher confidence)

  6. consistency = calculate_consistency_score(all_items, claim_numbers)
     (consistent numbers = higher confidence)

WEIGHTED FORMULA:
  confidence = 0.25 * total
             + 0.25 * balance
             + 0.15 * count_factor
             + 0.15 * avg_authority
             + 0.10 * diversity
             + 0.10 * consistency

WEIGHTS:
  - 25% arm strength (total)
  - 25% arm balance (gap between arms)
  - 15% item count
  - 15% average authority
  - 10% diversity
  - 10% consistency

CAPS:
  - All components clamped 0.0-1.0
  - Final confidence clamped 0.0-1.0
